[Unit]
Description=nethesis-insights

[Pod]
PodName=insights
# The only ports the pod publishes (node-exporter.container, outside the pod
# on the host network, adds 9100 for the metrics server only). Every backend
# listens on
# loopback inside the shared namespace and has no published port at all, which
# is what keeps the three unauthenticated operator UIs reachable only through
# Traefik's BasicAuth. deploy/host-firewall.sh does not change that: firewalld
# filters the host's own listeners, but netavark's DNAT sends a published
# container port around it (verified on the deployed host), so "not
# published" is still worth a great deal more than "bound to 127.0.0.1".
PublishPort=80:80
PublishPort=443:443

[Install]
WantedBy=multi-user.target
